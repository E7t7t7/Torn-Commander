sealed class Fake{
 public Fake(SlotsBatch _){}
 public bool CheckPage;
 public int Reads;public Action? OnRead;
 public int Clicks,Tokens=100,Owner=123,BoardOwner=123;
 public decimal Wallet=1000000,Payout=-1,ReceiptBet=1000;
 public bool Current=true,SuppressReceipt,Ambiguous,DispatchFalse,ThrowAfterClick,FailSave;
 public Action? AfterClick;public SlotReceipt? PendingReceipt;
 DateTime now=DateTime.UtcNow;
 public SlotsRuntime Runtime()=>new(){
  Current=()=>Current,Now=()=>now,
  Delay=ms=>{now=now.AddMilliseconds(ms);return Task.CompletedTask;},
  Account=()=>Task.FromResult(new SlotsAccount(Owner,Tokens,Wallet)),
  Board=()=>Task.FromResult(new SlotsBoard{Ok=true,Ready=true,Owner=BoardOwner,Tokens=Tokens}),
  Dispatch=p=>{
   if(DispatchFalse)return Task.FromResult(new SlotsDispatch{Dispatched=false,Note="No click"});
   Clicks++;Tokens--;var payout=Payout<0?(Clicks%2==0?5000:0):Payout;Wallet+=payout-1000;
   PendingReceipt=new("spin-"+Clicks,p.StartedUtc,ReceiptBet,payout);AfterClick?.Invoke();
   if(ThrowAfterClick)throw new IOException("Uncertain");
   return Task.FromResult(new SlotsDispatch{Dispatched=true});
  },
  Progress=CheckPage?_=>Task.FromResult(new SlotsProgress{Ok=true,Owner=Owner,Tokens=100,Activity=false}):null,
  Receipts=()=>{Reads++;OnRead?.Invoke();return Task.FromResult(SuppressReceipt||PendingReceipt==null?new List<SlotReceipt>():Ambiguous?new(){PendingReceipt,PendingReceipt with{Id="other"}}:new List<SlotReceipt>{PendingReceipt});},
  Acknowledge=_=>Task.CompletedTask,
  Save=()=>{if(FailSave)throw new IOException("Disk failure");}
 };
}
